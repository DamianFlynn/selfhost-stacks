def flat:
  ( ((.ffprobe.format.tags // {}) | to_entries | map({ key: (.key | ascii_downcase), value: { n: .key, v: (.value | tostring) } }) )
    + ( [ .ffprobe.streams[]? | select(.codec_type == "audio") ] | to_entries
        | map( (.key | tostring) as $i | ((.value.tags // {}) | to_entries
               | map({ key: ("stream" + $i + "." + (.key | ascii_downcase)), value: { n: ("stream" + $i + "." + .key), v: (.value | tostring) } }) ) )
        | add // [] ) ) | from_entries;
[ .[] | { k: (.audio_md5 // ""), p: (.source_path // ""), t: flat } ]
| group_by(.k) | map(select(length > 1)) as $g
| { multi_record_groups: ($g | length),
    multi_record_groups_multi_path: ($g | map(select((map(.p)|unique|length) > 1)) | length),
    identical: ($g | map(select((map(.t)|unique|length) == 1)) | length),
    divergent: ($g | map(select((map(.t)|unique|length) > 1)) | length),
    divergent_single_path: ($g | map(select((map(.t)|unique|length) > 1 and (map(.p)|unique|length) == 1)) | length),
    divergent_roots: ($g | map(select((map(.t)|unique|length) > 1)) | map(map(.p | split("/")[0:7] | join("/")) | unique | join(" + ")) | group_by(.) | map({roots: .[0], n: length}) | sort_by(-.n) | .[0:6]),
    divergent_fields_top: ($g | map(select((map(.t)|unique|length) > 1)) | map( . as $grp | ([$grp[].t | keys[]] | unique) as $ks | [ $ks[] as $k | select( ([$grp[] | (.t[$k].v // null)] | unique | length) > 1 ) | $k ] ) | flatten | group_by(.) | map({field: .[0], groups: length}) | sort_by(-.groups) | .[0:10])
  }
