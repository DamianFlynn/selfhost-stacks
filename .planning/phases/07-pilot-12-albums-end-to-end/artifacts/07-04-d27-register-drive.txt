  #!/usr/bin/env bash
  # d04-measure.sh <clone-dir> <label> [grep-binary]
  # Applies quick-health-check.sh's OWN D-04 regexes (extracted verbatim from the clone's copy of the
  # file, not retyped) and its OWN local pipeline to the block's own remote command's output.
  CLONE="$1"; LABEL="$2"; G="${3:-/usr/bin/grep}"
  QHC="$CLONE/scripts/quick-health-check.sh"
  eval "$(/usr/bin/grep -E '^[[:space:]]*D04_INV_RE=' "$QHC" | sed 's/^[[:space:]]*//')"
  eval "$(/usr/bin/grep -E '^[[:space:]]*D04_EXEMPT_RE=' "$QHC" | sed 's/^[[:space:]]*//')"
  D04_OUT=$(cd "$CLONE" && git grep -n -I -w -E -e 'beet' -e 'BEET[A-Z_]*' HEAD -- scripts stacks)
  D04_HITS=$(printf '%s\n' "$D04_OUT" | $G '^HEAD:')
  D04_RAW=$(printf '%s\n' "$D04_HITS" | $G -c '^HEAD:')
  D04_KEPT=$(printf '%s\n' "$D04_HITS" | $G -vE '^HEAD:[^:]*:[0-9]*:[[:space:]]*(#|//)')
  D04_STRIPPED=$(printf '%s\n' "$D04_KEPT" | $G -c '^HEAD:')
  D04_INVOKE_ALL=$(printf '%s\n' "$D04_KEPT" | $G -E "$D04_INV_RE")
  D04_INVOKE_DOC=$(printf '%s\n' "$D04_INVOKE_ALL" | $G '\.md:')
  D04_INVOKE_EXE=$(printf '%s\n' "$D04_INVOKE_ALL" | $G -v '\.md:' | $G '^HEAD:')
  D04_INVOKE_EXEMPT=$(printf '%s\n' "$D04_INVOKE_EXE" | $G -E "$D04_EXEMPT_RE")
  D04_INVOKE_ASSERT=$(printf '%s\n' "$D04_INVOKE_EXE" | $G -vE "$D04_EXEMPT_RE" | $G '^HEAD:')
  D04_N_ALL=$(printf '%s\n' "$D04_INVOKE_ALL" | $G -c '^HEAD:')
  D04_N_DOC=$(printf '%s\n' "$D04_INVOKE_DOC" | $G -c '^HEAD:')
  D04_N_EXE=$(printf '%s\n' "$D04_INVOKE_EXE" | $G -c '^HEAD:')
  D04_N_EXEMPT=$(printf '%s\n' "$D04_INVOKE_EXEMPT" | $G -c '^HEAD:')
  D04_N_ASSERT=$(printf '%s\n' "$D04_INVOKE_ASSERT" | $G -c '^HEAD:')
  echo "== $LABEL  (HEAD $(cd "$CLONE" && git rev-parse --short HEAD); matcher $G)"
  echo "D04_EXEMPT_RE in effect: $D04_EXEMPT_RE"
  echo "counts: raw=$D04_RAW comment-stripped=$D04_STRIPPED invocation-shaped=$D04_N_ALL  D04_N_EXE=$D04_N_EXE  D04_N_ASSERT=$D04_N_ASSERT  D04_N_EXEMPT=$D04_N_EXEMPT  D04_N_DOC=$D04_N_DOC"
  echo "-- D04_INVOKE_ASSERT ($D04_N_ASSERT):"; printf '%s\n' "$D04_INVOKE_ASSERT" | $G '^HEAD:' | cut -c1-200 | sed 's/^/   /'
  echo "-- D04_INVOKE_EXEMPT ($D04_N_EXEMPT):"; printf '%s\n' "$D04_INVOKE_EXEMPT" | $G '^HEAD:' | cut -c1-200 | sed 's/^/   /'
  echo "-- route-dj-album.sh lines in D04_INVOKE_ASSERT: $(printf '%s\n' "$D04_INVOKE_ASSERT" | $G -c '^HEAD:scripts/route-dj-album\.sh:')"
  echo "-- route-dj-album.sh lines in D04_INVOKE_EXEMPT: $(printf '%s\n' "$D04_INVOKE_EXEMPT" | $G -c '^HEAD:scripts/route-dj-album\.sh:')"
