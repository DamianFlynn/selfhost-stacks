manifest() {
  # $1 = dir, $2 = label
  say "MANIFEST $2 :: $1"
  timeout 120 env LC_ALL=C find "$1" -mindepth 1 -printf '%p\t%y\t%s\t%T@\n' | LC_ALL=C sort
  RC=$?
  if [ "$RC" -eq 124 ]; then echo "UNKNOWN — find exceeded its bound on $1"; exit 3; fi
  [ "$RC" -eq 0 ] || { echo "UNKNOWN — find rc=$RC on $1"; exit 3; }
}

count() {
  timeout 120 env LC_ALL=C find "$1" -mindepth 1 -printf 'x\n' | wc -l
  RC=$?
  if [ "$RC" -eq 124 ]; then echo "UNKNOWN — find exceeded its bound on $1" >&2; exit 3; fi
  [ "$RC" -eq 0 ] || { echo "UNKNOWN — find rc=$RC on $1" >&2; exit 3; }
}

hashes() {
  # $1 = dir. Prints "<sha256>  <path relative to $1>" sorted.
  timeout 600 env LC_ALL=C find "$1" -mindepth 1 -type f -print0 \
    | LC_ALL=C sort -z \
    | timeout 600 xargs -0 -r sha256sum \
    | sed "s#  $1/#  #" \
    | LC_ALL=C sort
  RC=$?
  if [ "$RC" -eq 124 ]; then echo "UNKNOWN — sha256sum exceeded its bound on $1"; exit 3; fi
  [ "$RC" -eq 0 ] || { echo "UNKNOWN — sha256 pipeline rc=$RC on $1"; exit 3; }
}

inodes() {
  # $1 = dir. Prints "<inode>\t<basename>" for top-level entries only.
  timeout 120 env LC_ALL=C find "$1" -mindepth 1 -maxdepth 1 -printf '%i\t%f\n' | LC_ALL=C sort -k2
  RC=$?
  [ "$RC" -eq 0 ] || { echo "UNKNOWN — inode find rc=$RC on $1"; exit 3; }
